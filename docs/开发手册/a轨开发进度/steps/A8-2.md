### 3.1 `keyword_service.py` — 完整文件

```python
"""关键词词库服务 — 四源汇聚 + LLM 四层分类 + CRUD。

四源：
  RawInputs → Enterprise.raw_inputs 拆词
  探针反推 → search_results.probe_query 提取
  SEO API  → MVP mock 假数据
  LLM生成  → gateway.chat() 补充长尾词

流程：
  generate_keywords() → 四源汇聚（去重）→ LLM 四层分类 → 批量写入
"""
import asyncio
import re
from collections import Counter
from typing import Any, Dict, List, Optional

from sqlalchemy.orm import Session
from sqlalchemy import or_

from app.core.llm.gateway import chat, simple_prompt
from app.core.logging_config import get_logger

logger = get_logger(__name__)

# 中文分词简易正则：2-15 个汉字/字母/数字组成的词
_WORD_RE = re.compile(r"[一-龥a-zA-Z0-9]{2,15}")


class KeywordService:
    """关键词词库服务 — 全静态方法，无状态。"""

    # ═══════════════════ CRUD ═══════════════════

    @staticmethod
    def list_keywords(
        db: Session,
        enterprise_id: int,
        layer: Optional[str] = None,
        source: Optional[str] = None,
        keyword_type: Optional[str] = None,
        pool_hint: Optional[str] = None,
        search: Optional[str] = None,
        page: int = 1,
        page_size: int = 50,
    ) -> Dict[str, Any]:
        """分页查询关键词列表。"""
        from app.models.strategy import Keyword

        q = db.query(Keyword).filter(Keyword.enterprise_id == enterprise_id)

        if layer:
            q = q.filter(Keyword.layer == layer)
        if source:
            q = q.filter(Keyword.source == source)
        if keyword_type:
            q = q.filter(Keyword.keyword_type == keyword_type)
        if pool_hint:
            q = q.filter(Keyword.pool_hint == pool_hint)
        if search:
            q = q.filter(Keyword.phrase.contains(search))

        total = q.count()
        rows = q.order_by(Keyword.id.desc()).offset((page - 1) * page_size).limit(page_size).all()

        return {
            "items": rows,
            "total": total,
            "page": page,
            "page_size": page_size,
        }

    @staticmethod
    def create_keyword(db: Session, enterprise_id: int, data: dict) -> Any:
        """手动添加关键词。"""
        from app.models.strategy import Keyword

        kw = Keyword(
            enterprise_id=enterprise_id,
            phrase=data.get("phrase", ""),
            layer=data.get("layer", ""),
            source=data.get("source", "手动"),
            lbs_tags=data.get("lbs_tags", []),
            keyword_type=data.get("keyword_type", "exact"),
            pool_hint=data.get("pool_hint", "core"),
        )
        db.add(kw)
        db.commit()
        db.refresh(kw)
        return kw

    @staticmethod
    def update_keyword(db: Session, enterprise_id: int, keyword_id: int, data: dict) -> Optional[Any]:
        """编辑关键词。"""
        from app.models.strategy import Keyword

        kw = db.query(Keyword).filter(
            Keyword.id == keyword_id,
            Keyword.enterprise_id == enterprise_id,
        ).first()
        if not kw:
            return None

        for field in ("phrase", "layer", "source", "lbs_tags", "keyword_type", "pool_hint", "status"):
            if field in data and data[field] is not None:
                setattr(kw, field, data[field])

        db.commit()
        db.refresh(kw)
        return kw

    @staticmethod
    def delete_keyword(db: Session, enterprise_id: int, keyword_id: int) -> bool:
        """删除关键词。"""
        from app.models.strategy import Keyword

        kw = db.query(Keyword).filter(
            Keyword.id == keyword_id,
            Keyword.enterprise_id == enterprise_id,
        ).first()
        if not kw:
            return False
        db.delete(kw)
        db.commit()
        return True

    @staticmethod
    def get_layer_summary(db: Session, enterprise_id: int) -> List[Dict[str, Any]]:
        """四层汇总 — 给 B 侧 E 区用。"""
        from app.models.strategy import Keyword

        rows = db.query(Keyword).filter(Keyword.enterprise_id == enterprise_id).all()
        layer_map: Dict[str, List[str]] = {}
        for r in rows:
            layer = r.layer or "未分类"
            layer_map.setdefault(layer, []).append(r.phrase)

        return [
            {"layer": layer, "count": len(phrases), "keywords": phrases}
            for layer, phrases in layer_map.items()
        ]

    # ═══════════════════ 四源汇聚 ═══════════════════

    @staticmethod
    def _from_raw_inputs(db: Session, enterprise_id: int) -> List[str]:
        """来源 1：从 Enterprise.raw_inputs 拆词。"""
        from app.models import Enterprise

        ent = db.query(Enterprise).filter(Enterprise.id == enterprise_id).first()
        if not ent or not ent.raw_inputs:
            return []

        # 简易拆词：按标点/换行切分 + 正则提取 2-15 字片段
        segments = re.split(r"[，。！？、\n,!.?\s]+", ent.raw_inputs)
        words = []
        for seg in segments:
            words.extend(_WORD_RE.findall(seg))
        return list(dict.fromkeys(words))  # 去重保序

    @staticmethod
    def _from_probes(db: Session, enterprise_id: int) -> List[str]:
        """来源 2：从 search_results.probe_query 提取关键词。"""
        from app.models.strategy import SearchResult

        rows = (
            db.query(SearchResult.probe_query)
            .filter(SearchResult.enterprise_id == enterprise_id)
            .distinct()
            .all()
        )

        words = []
        for (probe,) in rows:
            if probe:
                words.extend(_WORD_RE.findall(probe))
        return list(dict.fromkeys(words))

    @staticmethod
    def _from_seo(db: Session, enterprise_id: int) -> List[str]:
        """来源 3：SEO API — MVP mock 假数据。

        后续可替换为真实 SEO API（百度指数/5118 等）。
        """
        from app.models import Enterprise

        ent = db.query(Enterprise).filter(Enterprise.id == enterprise_id).first()
        city = ""
        if ent:
            # 尝试从 settings 或 stores 取城市信息
            city = (ent.settings or {}).get("city", "")

        # 美容行业通用高频词 + 地域词
        base = [
            "皮肤管理", "美容院", "护肤", "敏感肌", "痘痘肌",
            "补水保湿", "抗衰老", "美白淡斑", "深层清洁", "小气泡",
            "水光针", "光子嫩肤", "热玛吉", "超声刀", "果酸换肤",
            "VISIA检测", "皮肤管理培训", "日式皮肤管理", "韩式皮肤管理",
        ]
        if city:
            base.extend([
                f"{city}皮肤管理",
                f"{city}美容院推荐",
                f"{city}皮肤管理哪家好",
            ])
        return base

    @staticmethod
    async def _from_llm(enterprise_name: str, city: str, existing_words: List[str]) -> List[str]:
        """来源 4：LLM 补充长尾词。

        输入已有词列表，让 LLM 补充遗漏的长尾/场景词。
        """
        existing_text = "、".join(existing_words[:50]) if existing_words else "无"

        prompt = f"""你是美容行业 SEO 专家。以下是"{enterprise_name}"（{city}）已有的关键词：

{existing_text}

请补充 10-15 个遗漏的长尾关键词，覆盖：
1. 地域+服务组合（如"静安寺周末皮肤管理"）
2. 痛点场景（如"换季敏感肌急救"）
3. 人群细分（如"白领午休护肤"）
4. 价格咨询（如"皮肤管理一次多少钱"）

输出 JSON 数组：["词1", "词2", ...]
仅输出 JSON 数组，不要加 markdown。"""
        resp = await chat(simple_prompt(prompt, temperature=0.7, max_tokens=1000))
        if resp.ok:
            import json
            try:
                text = resp.content
                m = re.search(r"```(?:json)?\s*([\s\S]*?)```", text)
                if m:
                    text = m.group(1)
                parsed = json.loads(text.strip())
                return [p for p in parsed if isinstance(p, str)]
            except Exception:
                pass
        return []

    # ═══════════════════ LLM 四层分类 ═══════════════════

    @staticmethod
    async def _classify_by_llm(words: List[str], enterprise_name: str) -> List[Dict[str, Any]]:
        """用 LLM 将关键词分到四层 + 提取 LBS 标签。

        Returns:
            [{"phrase": "皮肤管理", "layer": "认知层", "lbs_tags": []}, ...]
        """
        if not words:
            return []

        words_text = "\n".join(f"{i+1}. {w}" for i, w in enumerate(words))

        prompt = f"""你是美容行业关键词分类专家。请将以下关键词分到四个层级：

层级定义：
- 认知层：品类/概念/技术名词（如"皮肤管理"、"VISIA检测"、"光子嫩肤"）
- 选型层：比价/地域选择/对比词（如"静安寺皮肤管理推荐"、"上海皮肤管理哪家好"）
- 痛点层：焦虑/犹豫/风险词（如"皮肤管理会不会越做越敏感"、"怕被推销办卡"）
- 场景层：时间/人群/具体需求词（如"白领午休皮肤管理"、"换季敏感肌急救"）

关键词列表（{enterprise_name}）：
{words_text}

请输出 JSON 数组，每个元素格式：
{{"phrase": "原词", "layer": "认知层|选型层|痛点层|场景层", "lbs_tags": ["地域标签"]}}

注意：
- lbs_tags 提取词中的地域信息（城市/区/商圈），没有则为空数组
- 仅输出 JSON 数组，不要加 markdown
- 每个输入词都要出现在输出中"""
        resp = await chat(simple_prompt(prompt, temperature=0.3, max_tokens=4000))
        if resp.ok:
            import json
            try:
                text = resp.content
                m = re.search(r"```(?:json)?\s*([\s\S]*?)```", text)
                if m:
                    text = m.group(1)
                parsed = json.loads(text.strip())
                return [p for p in parsed if isinstance(p, dict) and "phrase" in p]
            except Exception:
                pass

        # fallback：全部归为认知层
        return [{"phrase": w, "layer": "认知层", "lbs_tags": []} for w in words]

    # ═══════════════════ 全流程入口 ═══════════════════

    @staticmethod
    async def generate_keywords(
        db_factory,
        enterprise_id: int,
        enterprise_name: str = "",
        city: str = "",
    ) -> Dict[str, Any]:
        """四源汇聚 → 去重 → LLM 分类 → 批量写入。

        由 Celery 任务调用。
        """
        from app.models.strategy import Keyword

        db = db_factory()
        try:
            # 1. 四源汇聚
            raw_words = KeywordService._from_raw_inputs(db, enterprise_id)
            probe_words = KeywordService._from_probes(db, enterprise_id)
            seo_words = KeywordService._from_seo(db, enterprise_id)

            logger.info(
                "[keyword] sources raw=%s probes=%s seo=%s",
                len(raw_words), len(probe_words), len(seo_words),
            )

            # 2. 合并 + 去重（保留首次来源）
            seen: Dict[str, str] = {}  # phrase → source
            for w in raw_words:
                seen[w] = "RawInputs"
            for w in probe_words:
                if w not in seen:
                    seen[w] = "探针反推"
            for w in seo_words:
                if w not in seen:
                    seen[w] = "SEO API"

            # 3. LLM 补充长尾词
            llm_words = await KeywordService._from_llm(
                enterprise_name, city, list(seen.keys()),
            )
            for w in llm_words:
                if w not in seen:
                    seen[w] = "LLM生成"

            logger.info("[keyword] merged unique=%s llm_new=%s", len(seen), len(llm_words))

            # 4. LLM 四层分类
            classified = await KeywordService._classify_by_llm(
                list(seen.keys()), enterprise_name,
            )

            # 5. 批量写入（跳过已存在的 phrase）
            existing = {
                r.phrase
                for r in db.query(Keyword.phrase).filter(
                    Keyword.enterprise_id == enterprise_id,
                    Keyword.phrase.in_([c["phrase"] for c in classified]),
                ).all()
            }

            inserted = 0
            skipped = 0
            for item in classified:
                phrase = item["phrase"]
                if phrase in existing:
                    skipped += 1
                    continue
                kw = Keyword(
                    enterprise_id=enterprise_id,
                    phrase=phrase,
                    layer=item.get("layer", ""),
                    source=seen.get(phrase, "LLM生成"),
                    lbs_tags=item.get("lbs_tags", []),
                    keyword_type="exact",
                    pool_hint="core",
                    status="active",
                )
                db.add(kw)
                inserted += 1

            db.commit()
            logger.info("[keyword] generate done inserted=%s skipped=%s", inserted, skipped)

            return {
                "total_sources": len(seen),
                "inserted": inserted,
                "skipped": skipped,
                "sources_breakdown": {
                    "raw_inputs": len(raw_words),
                    "probes": len(probe_words),
                    "seo": len(seo_words),
                    "llm": len(llm_words),
                },
            }
        finally:
            db.close()
```
