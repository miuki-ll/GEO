## 3. 改动清单

### 3.1 改动 1：`views/settings/Index.vue` — 接真实 API

**文件**：`frontend/src/views/settings/Index.vue`

**改动要点**：

#### A. 企业信息 tab

```typescript
import { ref, reactive, onMounted } from 'vue'
import { ElMessage } from 'element-plus'
import { getEnterprise, updateEnterprise, listMembers, updateMember } from '@/api/enterprise'
import type { EnterpriseUpdate } from '@/api/enterprise'

const tab = ref('profile')

// ── 企业信息 ──
const profileLoading = ref(false)
const profile = reactive({
  name: '',
  industry: 'beauty_local',
  license: '',
  contact_name: '',
  phone: '',
  email: '',
})

async function loadProfile() {
  profileLoading.value = true
  try {
    const res = await getEnterprise()
    const d = res.data
    if (d) {
      profile.name = d.name || ''
      profile.industry = d.industry || 'beauty_local'
      profile.license = d.license_no || ''
      profile.contact_name = d.contact_name || ''
      profile.phone = d.contact_phone || ''
      profile.email = d.contact_email || ''
    }
  } catch {
    // 加载失败保持默认
  } finally {
    profileLoading.value = false
  }
}

async function save() {
  try {
    const data: EnterpriseUpdate = {
      name: profile.name,
      industry: profile.industry,
      contact_name: profile.contact_name,
      contact_phone: profile.phone,
      contact_email: profile.email,
    }
    await updateEnterprise(data)
    ElMessage.success('已保存')
  } catch (e: any) {
    ElMessage.error(e?.response?.data?.detail || '保存失败')
  }
}
```

#### B. 团队成员 tab

```typescript
// ── 团队成员 ──
const members = ref<any[]>([])
const membersLoading = ref(false)

async function loadMembers() {
  membersLoading.value = true
  try {
    const res = await listMembers({ page_size: 50 })
    members.value = (res.data || []).map((m: any) => ({
      ...m,
      role_cn: { owner: 'Owner', admin: '管理员', editor: '编辑', viewer: '只读' }[m.role] || m.role,
      join: m.created_at || '',
    }))
  } finally {
    membersLoading.value = false
  }
}

// 切换 tab 时加载
import { watch } from 'vue'
watch(tab, (v) => {
  if (v === 'profile') loadProfile()
  else if (v === 'members') loadMembers()
})

onMounted(() => { loadProfile() })
```

> ⚠️ LLM 引擎配置和通知 tab 保持现有占位 UI，不接 API。

### 3.2 改动 2：`router/index.ts` — 路由守卫加入驻状态判断

**文件**：`frontend/src/router/index.ts`

**位置**：`router.beforeEach` 中，`userStore.fetchUser()` 之后

**改什么**：判断用户是否已完成入驻，未入驻则跳转 /onboarding

```typescript
// 在 router.beforeEach 中，fetchUser 成功后追加：
if (!userStore.user) {
  userStore.logout()
  return next({ name: 'Login' })
}

// 新增：未入驻用户强制跳转入驻页
// 判断方式：检查 enterprise 是否有 brand（A5 入驻后会写入 Brand）
// 或者检查 enterprise.settings.onboarding_done 标记
const isOnboarded = userStore.user.enterprise?.settings?.onboarding_done
  || userStore.user.enterprise?.has_brand  // 后端 getCurrentUser 返回的标记
if (!isOnboarded && to.name !== 'Onboarding' && to.name !== 'Settings') {
  return next({ name: 'Onboarding' })
}

next()
```

> ⚠️ 注意：需要确认后端 `GET /user/enterprise/profile` 或 `getCurrentUser` 返回的数据中是否有入驻完成标记。如果没有，需要：
> - 方案 A：后端加一个 `onboarding_done` 字段到 Enterprise.settings
> - 方案 B：前端调 `getEnterprise()` 检查是否有 brand 数据
>
> **推荐方案 A**：A5 入驻完成后在 `_write_side_effects` 中设置 `enterprise.settings["onboarding_done"] = True`。
