## 2. Login.vue 改动

**文件**: `frontend/src/views/Login.vue`

### 2.1 模板改动

- 删除底部的 MVP 提示 `<el-alert>`（第 24-26 行）

### 2.2 Script 改动

- 去掉 `form` 的默认值 `demo@geo.test / demo123456`
- `handleLogin()` 改为调 `userStore.login(form.email, form.password)`
- 加错误处理：`401` → "邮箱或密码错误"，网络异常 → "网络错误，请稍后重试"

```typescript
const form = reactive({
  email: '',
  password: '',
})

async function handleLogin() {
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) return
  loading.value = true
  try {
    await userStore.login(form.email, form.password)
    ElMessage.success('登录成功')
    const redirect = (route.query.redirect as string) || '/outcomes'
    router.push(redirect)
  } catch (e: any) {
    if (e?.response?.status === 401) {
      ElMessage.error('邮箱或密码错误')
    } else {
      ElMessage.error('网络错误，请稍后重试')
    }
  } finally {
    loading.value = false
  }
}
```

> **说明**: `userStore.login()` 已在 `stores/user.ts:23-29` 实现完毕 — 调 `apiLogin` → 存 token + user → 持久化到 localStorage。无需改动 store。

## 3. Register.vue 改动

**文件**: `frontend/src/views/Register.vue`

### 3.1 模板改动

- 给表单加 `ref="formRef"`、`:model`、`:rules`、`@submit.prevent`
- 每个 `el-form-item` 加 `prop` 属性对应字段名
- 提交按钮加 `:loading`

### 3.2 Script 改动

- 引入表单校验相关 API：`ref`, `FormInstance`, `FormRules`, `ElMessage`
- 引入 `useUserStore` 和 `router`
- 加 form 校验规则：email 格式、password >= 6、enterprise/name 必填
- `handleRegister()` 调 `api/auth.ts` 的 `register()` → 成功调 `userStore.login()` → 跳转 `/onboarding`
- 加 loading + 错误处理（邮箱已注册、企业名重复 → 400 提示）

```typescript
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage, FormInstance, FormRules } from 'element-plus'
import { register } from '@/api/auth'
import { useUserStore } from '@/stores/user'

const router = useRouter()
const userStore = useUserStore()
const formRef = ref<FormInstance>()
const loading = ref(false)

const form = reactive({
  enterprise: '',
  name: '',
  email: '',
  password: '',
})

const rules: FormRules = {
  enterprise: [{ required: true, message: '请输入企业名称', trigger: 'blur' }],
  name: [{ required: true, message: '请输入联系人姓名', trigger: 'blur' }],
  email: [
    { required: true, message: '请输入邮箱', trigger: 'blur' },
    { type: 'email', message: '邮箱格式不正确', trigger: 'blur' },
  ],
  password: [
    { required: true, message: '请输入密码', trigger: 'blur' },
    { min: 6, message: '密码至少 6 位', trigger: 'blur' },
  ],
}

async function handleRegister() {
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) return
  loading.value = true
  try {
    await register({
      email: form.email,
      password: form.password,
      full_name: form.name,
      enterprise_name: form.enterprise,
    })
    await userStore.login(form.email, form.password)
    ElMessage.success('注册成功')
    router.push('/onboarding')
  } catch (e: any) {
    const msg = e?.response?.data?.detail || '注册失败，请稍后重试'
    ElMessage.error(msg)
  } finally {
    loading.value = false
  }
}
```

> **注意**: `register()` 返回后不直接含 token（与 login 不同），需要额外调一次 `userStore.login()` 完成自动登录。
